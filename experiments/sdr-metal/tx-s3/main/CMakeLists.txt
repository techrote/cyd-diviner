idf_component_register(SRCS "tx_metal_s3.c" REQUIRES esp_wifi esp_phy esp_driver_usb_serial_jtag esp_driver_tsens esp_timer esp_hw_support app_update)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/tx_metal_s3.c" "${CMAKE_CURRENT_LIST_DIR}/metal_words.h" "${CMAKE_CURRENT_LIST_DIR}/fast_schedule.h")
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}" OUTPUT_VARIABLE source_sha OUTPUT_STRIP_TRAILING_WHITESPACE)
execute_process(COMMAND git diff --quiet WORKING_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}" RESULT_VARIABLE source_dirty)
file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/tx_metal_s3.c" source_digest)
file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/metal_words.h" words_digest)
file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/fast_schedule.h" fast_digest)
target_compile_definitions(${COMPONENT_LIB} PRIVATE SDR_SOURCE_SHA="${source_sha}" SDR_SOURCE_DIRTY=${source_dirty} SDR_SOURCE_DIGEST="${source_digest}" SDR_WORDS_DIGEST="${words_digest}" SDR_FAST_DIGEST="${fast_digest}")
